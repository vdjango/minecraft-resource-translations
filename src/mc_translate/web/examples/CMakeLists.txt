cmake_minimum_required(VERSION 3.20)
project(translation_examples LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
find_package(CURL REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(cJSON CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
add_executable(translation_c client.c)
target_link_libraries(translation_c PRIVATE CURL::libcurl OpenSSL::Crypto cjson)
add_executable(translation_cpp client.cpp)
target_link_libraries(translation_cpp PRIVATE CURL::libcurl OpenSSL::Crypto nlohmann_json::nlohmann_json)
